cmake_minimum_required(VERSION 3.20)
project(th12_multiplayer_runtime LANGUAGES CXX)

option(TH12_BUILD_SDL_HOST "Build the shared SDL3 / GLES 3 platform library" OFF)
set(TH12_MULTIPLAYER ON)
if(TH12_BUILD_SDL_HOST)
  enable_language(C)
endif()

file(GLOB TH12_GAME_SOURCES CONFIGURE_DEPENDS src/game/*.cpp)
add_library(th12_runtime STATIC ${TH12_GAME_SOURCES} src/wasm/Exports.cpp src/wasm/MultiplayerExports.cpp)
target_compile_features(th12_runtime PUBLIC cxx_std_20)
target_include_directories(th12_runtime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
set_target_properties(th12_runtime PROPERTIES CXX_EXTENSIONS OFF)
option(TH12_ENABLE_THPRAC "Build the purple ImGui practice adapter" OFF)
set(EAGLER_COMMON_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../common" CACHE PATH "Shared Eagler sources")
target_compile_definitions(th12_runtime PUBLIC TH12_MULTIPLAYER=1)
target_include_directories(th12_runtime PUBLIC "${EAGLER_COMMON_ROOT}/include")
file(GLOB TH12_MULTIPLAYER_SOURCES CONFIGURE_DEPENDS src/multiplayer/*.cpp)
file(GLOB TH12_NETPLAY_SOURCES CONFIGURE_DEPENDS "${EAGLER_COMMON_ROOT}/src/netplay/*.cpp")
target_sources(th12_runtime PRIVATE ${TH12_MULTIPLAYER_SOURCES} ${TH12_NETPLAY_SOURCES})
target_include_directories(th12_runtime PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/multiplayer")
if(TH12_BUILD_SDL_HOST)
  include(portable/sdl/Host.cmake)
endif()
if(TH12_ENABLE_THPRAC)
  set(TH12_IMGUI "${EAGLER_COMMON_ROOT}/third_party/imgui")
  target_compile_definitions(th12_runtime PUBLIC TH12_ENABLE_THPRAC=1 IMGUI_DISABLE_WIN32_FUNCTIONS)
  if(TARGET Freetype::Freetype)
    target_link_libraries(th12_runtime PRIVATE Freetype::Freetype)
  elseif(TARGET freetype)
    target_link_libraries(th12_runtime PRIVATE freetype)
  elseif(FREETYPE_SOURCE_DIR)
    set(FT_DISABLE_ZLIB ON CACHE BOOL "" FORCE)
    set(FT_DISABLE_BZIP2 ON CACHE BOOL "" FORCE)
    set(FT_DISABLE_PNG ON CACHE BOOL "" FORCE)
    set(FT_DISABLE_HARFBUZZ ON CACHE BOOL "" FORCE)
    set(FT_DISABLE_BROTLI ON CACHE BOOL "" FORCE)
    add_subdirectory("${FREETYPE_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/freetype" EXCLUDE_FROM_ALL)
    target_link_libraries(th12_runtime PRIVATE freetype)
  else()
    find_package(Freetype REQUIRED)
    target_link_libraries(th12_runtime PRIVATE Freetype::Freetype)
  endif()
  target_sources(th12_runtime PRIVATE "${TH12_IMGUI}/imgui_freetype.cpp")
  target_include_directories(th12_runtime PUBLIC "${EAGLER_COMMON_ROOT}/include" "${TH12_IMGUI}")
  target_sources(th12_runtime PRIVATE src/wasm/PracticeUi.cpp "${TH12_IMGUI}/imgui.cpp" "${TH12_IMGUI}/imgui_draw.cpp" "${TH12_IMGUI}/imgui_tables.cpp" "${TH12_IMGUI}/imgui_widgets.cpp")
endif()
