cmake_minimum_required(VERSION 3.20)
project(th12_mp_profile LANGUAGES CXX)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(TH12_ENABLE_THPRAC OFF CACHE BOOL "" FORCE)
get_filename_component(EAGLER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${EAGLER_ROOT}/ports/th12-mp" core)
target_compile_definitions(th12_runtime PUBLIC TH12_ENABLE_THPRAC=1)
target_compile_options(th12_runtime PUBLIC /utf-8 /fp:strict)
add_executable(profile native-profile.cpp)
target_link_libraries(profile PRIVATE th12_runtime)
