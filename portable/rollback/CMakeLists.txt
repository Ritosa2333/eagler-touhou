cmake_minimum_required(VERSION 3.20)
project(th15_rollback_checks LANGUAGES CXX)
enable_testing()
get_filename_component(TITLE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
get_filename_component(DEFAULT_COMMON "${TITLE_ROOT}/../../common" ABSOLUTE)
set(EAGLER_COMMON_ROOT "${DEFAULT_COMMON}" CACHE PATH "Shared Eagler sources")
file(GLOB GAME_SOURCES CONFIGURE_DEPENDS "${TITLE_ROOT}/web/cpp/game/*.cpp")
add_library(th15_rollback_game STATIC ${GAME_SOURCES}
    "${TITLE_ROOT}/web/cpp/multiplayer/BattleFrameState.cpp"
    "${TITLE_ROOT}/web/cpp/multiplayer/StageFrameState.cpp"
    "${TITLE_ROOT}/web/cpp/multiplayer/AnimationState.cpp"
    "${TITLE_ROOT}/web/cpp/multiplayer/FrameState.cpp"
    "${TITLE_ROOT}/web/cpp/multiplayer/ItemFrameState.cpp"
    "${TITLE_ROOT}/web/cpp/multiplayer/EnemyFrameState.cpp")
target_compile_features(th15_rollback_game PUBLIC cxx_std_17)
target_compile_definitions(th15_rollback_game PUBLIC TH15_MULTIPLAYER=1 TH_NATIVE_PLATFORM=1 TH_ENABLE_THPRAC=0)
target_include_directories(th15_rollback_game PUBLIC "${TITLE_ROOT}/web/cpp" "${EAGLER_COMMON_ROOT}/include")
if(MSVC)
    target_compile_options(th15_rollback_game PUBLIC /utf-8 /fp:strict)
else()
    target_compile_options(th15_rollback_game PUBLIC -ffp-contract=off -fno-strict-aliasing)
endif()
add_executable(th15_animation_rollback animation-state.cpp)
target_link_libraries(th15_animation_rollback PRIVATE th15_rollback_game)
add_test(NAME th15_animation_rollback COMMAND th15_animation_rollback)
add_executable(th15_frame_rollback frame-state.cpp)
target_link_libraries(th15_frame_rollback PRIVATE th15_rollback_game)
add_test(NAME th15_frame_rollback COMMAND th15_frame_rollback)
add_executable(th15_visible_fps visible-fps.cpp)
target_compile_features(th15_visible_fps PUBLIC cxx_std_17)
add_test(NAME th15_visible_fps COMMAND th15_visible_fps)
add_executable(th15_frame_effects frame-effects.cpp)
target_compile_features(th15_frame_effects PUBLIC cxx_std_17)
add_test(NAME th15_frame_effects COMMAND th15_frame_effects)
add_executable(th15_enemy_rollback enemy-state.cpp)
target_link_libraries(th15_enemy_rollback PRIVATE th15_rollback_game)
add_test(NAME th15_enemy_rollback COMMAND th15_enemy_rollback)
add_executable(th15_laser_rollback laser-state.cpp)
target_link_libraries(th15_laser_rollback PRIVATE th15_rollback_game)
add_executable(th15_battle_rollback battle-state.cpp)
target_link_libraries(th15_battle_rollback PRIVATE th15_rollback_game)
set(TH15_TEST_ARCHIVE "" CACHE FILEPATH "Legally supplied TH15 1.00b archive")
if(EXISTS "${TH15_TEST_ARCHIVE}")
    add_test(NAME th15_laser_rollback COMMAND th15_laser_rollback "${TH15_TEST_ARCHIVE}")
    add_test(NAME th15_battle_rollback COMMAND th15_battle_rollback "${TH15_TEST_ARCHIVE}")
endif()
file(GLOB NETWORK_SOURCES "${EAGLER_COMMON_ROOT}/src/netplay/*.cpp")
list(FILTER NETWORK_SOURCES EXCLUDE REGEX "NetplayInput\\.cpp$")
add_library(th15_rollback_network STATIC ${NETWORK_SOURCES}
    "${TITLE_ROOT}/web/cpp/multiplayer/NetplayRuntime.cpp"
    "${TITLE_ROOT}/web/cpp/multiplayer/SessionSetup.cpp"
    "${TITLE_ROOT}/web/cpp/multiplayer/ReplayArchive.cpp"
    "${TITLE_ROOT}/web/cpp/multiplayer/InputLanes.cpp")
target_compile_features(th15_rollback_network PUBLIC cxx_std_17)
target_compile_definitions(th15_rollback_network PUBLIC TH15_MULTIPLAYER=1 TH_NATIVE_PLATFORM=1)
target_include_directories(th15_rollback_network PUBLIC "${TITLE_ROOT}/web/cpp/multiplayer" "${EAGLER_COMMON_ROOT}/include")
if(MSVC)
    target_compile_options(th15_rollback_network PUBLIC /utf-8 /fp:strict)
endif()
add_executable(th15_network_rollback ../multiplayer/network-protocol.cpp)
target_link_libraries(th15_network_rollback PRIVATE th15_rollback_network)
add_test(NAME th15_network_rollback COMMAND th15_network_rollback)
add_executable(th15_shared_startup "${EAGLER_COMMON_ROOT}/tests/adonis-startup-test.cpp")
target_link_libraries(th15_shared_startup PRIVATE th15_rollback_network)
if(MSVC)
    target_compile_options(th15_shared_startup PRIVATE /UNDEBUG)
else()
    target_compile_options(th15_shared_startup PRIVATE -UNDEBUG)
endif()
add_test(NAME th15_shared_startup COMMAND th15_shared_startup)
