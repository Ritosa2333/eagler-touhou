cmake_minimum_required(VERSION 3.25)
project(thkkd VERSION 0.1.0 LANGUAGES CXX)
if(NOT MSVC OR NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This desktop build requires Visual Studio and Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
add_executable(thkkd WIN32 src/platform/windows/Main.cpp src/platform/windows/WindowsHost.cpp src/platform/windows/DisplaySettings.cpp)
add_executable(thkkd-config WIN32 src/platform/windows/ConfigMain.cpp src/platform/windows/DisplaySettings.cpp)
target_compile_definitions(thkkd-config PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(thkkd-config PRIVATE /utf-8 /W3)
target_link_libraries(thkkd-config PRIVATE user32 gdi32)
set_target_properties(thkkd-config PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
file(GLOB DRAWING_SOURCES CONFIGURE_DEPENDS src/render/*.cpp)
target_sources(thkkd PRIVATE ${DRAWING_SOURCES})
set_source_files_properties(${DRAWING_SOURCES} PROPERTIES COMPILE_OPTIONS /wd4102)
target_include_directories(thkkd PRIVATE src third_party/dxlib)
target_compile_definitions(thkkd PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(thkkd PRIVATE /utf-8 /fp:strict /W3 /bigobj)
target_link_directories(thkkd PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/dxlib")
set_target_properties(thkkd PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
install(TARGETS thkkd thkkd-config RUNTIME DESTINATION .)
install(DIRECTORY assets DESTINATION .)
install(FILES docs/PLAY.txt DESTINATION . RENAME README.txt)
install(FILES THIRD_PARTY.md DESTINATION .)
install(FILES third_party/dxlib/DxLib.txt third_party/dxlib/VERSION.json DESTINATION third_party/dxlib)
