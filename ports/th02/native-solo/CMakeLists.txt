# FindZLIB's static preference can still return import libraries on Windows.
set(SOEW_ZLIB_TARGET "")
if(NOT FETCHCONTENT_SOURCE_DIR_ZLIB)
    block(PROPAGATE SOEW_ZLIB_TARGET)
        set(ZLIB_USE_STATIC_LIBS ON)
        find_package(ZLIB QUIET MODULE)
        if(ZLIB_FOUND)
            # Older FindZLIB modules assume a three-component version, but zlib 1.4 uses two.
            file(STRINGS "${ZLIB_INCLUDE_DIR}/zlib.h" zlib_version_line REGEX "^#define ZLIB_VERSION ")
            if(zlib_version_line MATCHES "\"([0-9]+([.][0-9]+)+)" AND
               CMAKE_MATCH_1 VERSION_GREATER_EQUAL 1.4)
                set(SOEW_ZLIB_TARGET ZLIB::ZLIB)
                foreach(config "" _RELEASE _DEBUG)
                    get_target_property(library ZLIB::ZLIB "IMPORTED_LOCATION${config}")
                    if(NOT library)
                        continue()
                    endif()
                    get_filename_component(filename "${library}" NAME)
                    string(TOLOWER "${filename}" filename)
                    # Accept conventional static archives, not .dll.a or ambiguous .lib names.
                    if(filename MATCHES "[.]dll[.]a$" OR
                       NOT filename MATCHES "([.]a$|^(lib)?(zs|zlibstatic|zlibstat)d?[.]lib$)")
                        set(SOEW_ZLIB_TARGET "")
                        break()
                    endif()
                endforeach()
            endif()
        endif()
    endblock()
endif()

function(soew_late_checkpoint stage mode size digest)
    set(key "SOEW_PRIVATE_${stage}_${mode}_CHECKPOINT")
    set(${key} "" CACHE FILEPATH "Verified ${stage} PRE BOSS audio checkpoint; private builds only")
    if(${key})
        if(SOEW_PRIVATE_SLIM_MIDI_CHECKPOINTS AND mode STREQUAL "MIDI")
            set(digest "${soew_slim_${key}}")
        endif()
        file(SIZE "${${key}}" actual_size)
        file(SHA256 "${${key}}" actual_digest)
        if(NOT actual_size EQUAL size OR NOT actual_digest STREQUAL digest)
            message(FATAL_ERROR "Private Extra ${mode} checkpoint does not match the verified artifact")
        endif()
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${${key}}")
        file(READ "${${key}}" bytes HEX)
        string(REGEX REPLACE "(..)" "std::byte{0x\\1}," entries "${bytes}")
        file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${key}.inc" CONTENT "${entries}\n")
        target_include_directories(soew_audio_preview PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
        target_compile_definitions(soew_audio_preview PRIVATE "${key}=1")
        target_link_libraries(soew_audio_preview PRIVATE bcrypt)
    endif()
endfunction()

    function(soew_catalog_checkpoint stage measure fm_size fm_digest midi_size midi_digest)
        math(EXPR fm_id "1000 + 512 * ${stage} + 2 * ${measure}")
        math(EXPR midi_id "${fm_id} + 1")
        foreach(mode fm midi)
            if(mode STREQUAL "fm")
                set(folder fm-fast)
                set(ext fmc)
            else()
                set(folder practice-midi-fast)
                set(ext mmdc)
                if(SOEW_PRIVATE_SLIM_MIDI_CHECKPOINTS)
                    set(midi_digest "${soew_slim_catalog_${stage}_${measure}}")
                endif()
            endif()
            set(name "catalog-STAGE${stage}-m${measure}.${ext}")
            set(input "${SOEW_PRIVATE_CHECKPOINT_CATALOG}/${folder}/${name}")
            if(NOT EXISTS "${input}" OR IS_DIRECTORY "${input}")
                message(FATAL_ERROR "Private checkpoint catalog entry is missing: ${input}")
            endif()
            file(SIZE "${input}" actual_size)
            file(SHA256 "${input}" actual_digest)
            if(NOT actual_size EQUAL ${mode}_size OR NOT actual_digest STREQUAL ${mode}_digest)
                message(FATAL_ERROR "Private checkpoint catalog entry does not match its pinned size and SHA256: ${input}")
            endif()
            set(staged "${catalog_dir}/${name}")
            configure_file("${input}" "${staged}" COPYONLY)
            list(APPEND catalog_assets "${staged}")
            if(MINGW)
                # As with the icon, run windres next to the staged files.
                set(resource_path "${name}")
            else()
                set(resource_path "${staged}")
            endif()
            string(APPEND catalog_resources "${${mode}_id} RCDATA \"${resource_path}\"\n")
        endforeach()
        string(APPEND catalog_entries "{${measure}, ${fm_id}, ${midi_id}},\n")

        set(catalog_resources "${catalog_resources}" PARENT_SCOPE)
